const getEntityGuarantor = (entity) => {
  if (entity) {
    let guarantor = null;

    if (entity.entityTypeId === 'process') {
      guarantor = entity.processOwner;
    }

    if (entity.entityTypeId === 'project') {
      guarantor = entity.projectManager;
    }

    if (entity.entityTypeId === 'risk') {
      guarantor = entity.riskOwner || entity.riskAuthor;
    }

    return guarantor || entity.guarantor || null;
  }

  return null;
};

  getUserIdList.push(rtp.ownerId);
  const riskEntityIdMap = {};
  
  const fileList = await getGraphql({

        id parentId name entityId parentEntityIdList

      }

    `,
    variables: {
      input: {
        filter: {
          dateValidFrom: {
            $gte: rtp.validFrom,
          },
          dateValidTo: {
            $lte: rtp.validTo,
          },
          schemaId: rtp.evaluationSchemaIdList[0],
          parentEntityIdList: {
            $raw: `this && array['${rtp.riskIdList.join("','")}']::varchar[]`,

      targetEvaluationByParentIdMap[evaluation.parentEntityIdList[0]] = evaluation;
      currentEvaluationByParentIdMap[evaluation.parentEntityIdList[0]] = evaluationMap[evaluation.parentId];

        measure.evaluationCurrent = evaluation;
      }

        measure.evaluationTarget = evaluation;
      }

      (map[measure.sourceId] = map[measure.sourceId] || []).push(measure);

      Object.values(measure.relatedData).filter(x => riskEntityIdMap[x.id]).forEach((x) => {
        if (x.entityTypeId === 'risk') {
          (map[x.id] = map[x.id] || []).push(measure);
        }
      });
    }

    return map;
  }, {});
  
  const planMeasureMap = {};
  
  rtp.relationList.filter(x => x.relationTypeId === 'include').forEach((relation) => {

        planMeasureMap[measure.id] = measure;

  let measureCostsActual = null;

  let measureCostsTotalUnknown = null;
  let measureCostsUnknown = null;
  
  planMeasureList.forEach((measure) => {
    if (planMeasureEntityMap[measure.targetId]) {          
      activeMeasureMap[measure.id] = measure;
      measureCostsTotal = measureCostsTotal || 0;

      if (measure.state === 'toApprove') {
        measureCostsTotalUnknown = measureCostsTotalUnknown || 0;
        measureCostsTotalUnknown += measure.costs.total;

      if (measure.costs && isNumeric(measure.costs.value)) {
        measureCosts = measureCosts || 0;
        measureCosts += measure.costs.value;

      if (measure.costs && isNumeric(measure.costs.valueActual)) {

        measureCostsActual += measure.costs.valueActual;

            note: qa.answerNote,
          });
        });
      }
      
      if (planMeasureEntityMap[data.id] && rtpDataSet.data.restrictionDataIdList.includes(x.id)) {
        dataMap.measureDataSetMap[x.id] = data;
      }    
    });
  }
  
  const costs = isNumeric(measureCostsActual) ? measureCostsActual : measureCosts;  
  let costsDifference = null;
  
  if (measureCostsTotal !== null && costs !== null) {

  }

      ...entity,

      evaluationTarget: targetEvaluationByParentIdMap[entity.id] || null,
      guarantorId: getEntityGuarantor(entity) || null,
      measureList: measureListByRiskEntityIdMap[entity.id] || [],
      riskTreatmentStrategy: (targetEvaluationByParentIdMap[entity.id]?.treatmentStrategy) || null,
      approvalWorkflow: approvalWorkflowMap[entity.id],
      isApprovalCompleted: !!(approvalWorkflowMap[entity.id]?.isCompleted && approvalWorkflowMap[entity.id].metadata.dateCompleted),
      workflowOwnerId: approvalWorkflowMap[entity.id]?.ownerId || null,
      workflowStateId: approvalWorkflowMap[entity.id]?.stateId || null,
      workflowState: stateMap[approvalWorkflowMap[entity.id]?.stateId]?.nameTranslated || null,
      created: getZotyHelpers('date', entity.createdAt, 'format("l, LT")'),
      updated: getZotyHelpers('date', entity.updatedAt, 'format("l, LT")'),
    };
    
    if (riskRecord.riskTreatmentStrategy) {
      riskRecord.riskTreatmentStrategy = i18n(`riskTreatmentStrategies.${riskRecord.riskTreatmentStrategy}`);
    }

    riskRecord.approvalOwnerId = riskRecord.guarantorId || riskRecord.ownerId;
    rtp.riskEntityList.push(riskRecord);
    getUserIdList.push(entity.authorId);
    getUserIdList.push(entity.ownerId);
    getUserIdList.push(riskRecord.guarantorId);
    getUserIdList.push(entity.approvalOwnerId);
    getUserIdList.push(entity.workflowOwnerId);
  });
  
  getUserIdList = getUserIdList.filter(x => x);
  const userList = await getUserListRequest({ filter: { id: getUserIdList, }, });

  rtp.owner = fillUser(rtp.ownerId);
  rtp.author = fillUser(rtp.authorId);
  
  rtp.riskEntityList.forEach(entity => {
    entity.owner = fillUser(entity.ownerId);
    entity.author = fillUser(entity.authorId);
    entity.guarantor = fillUser(entity.guarantorId);
    entity.approvalOwner = fillUser(entity.approvalOwnerId);
    entity.workflowOwner = fillUser(entity.workflowOwnerId);
  });
  
  rtp.fileList.forEach(file => {
    file.created = getZotyHelpers('date', file.createdAt, 'format("l, LT")'),
    file.author = fillUser(file.authorId);
  });
  
  dataMap = {
    key: rtp.key,
    riskIdList: rtp.riskIdList,
    activeMeasureMap,
    currentEvaluationByParentIdMap,
    targetEvaluationByParentIdMap,
    costsActualMissing: measureCostsUnknown,
    costsActualTotal: measureCostsActual - (measureCostsUnknown || 0),
    costs,
    costsTotal: measureCostsTotal,
    costsDifference,
    riskEntityList: rtp.riskEntityList,
    evaluationList: rtp.evaluationList,
    measureDataSetMap: dataMap.measureDataSetMap || {},
    measureEntityIdList: Object.values(planMeasureEntityMap).map(x => x.id),
    measureEntityList: Object.values(planMeasureEntityMap),
    measureEntityMap: planMeasureEntityMap,
    measureList: planMeasureList,
    measureListByRiskEntityIdMap,
    ownerId: rtp.ownerId,
    schemaId: rtp.evaluationSchemaIdList[0],
  };
}
