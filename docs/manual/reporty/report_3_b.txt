// Vrati hodnoceni pro Riziko ECRS1YCtGEXx.
let evaluationList = await getGraphql({
  query: `query getEvaluationList($input: EvaluationListInputType) {
    evaluationListQuery(input: $input) {
      id parentId name entityId parentEntityIdList
      score { customName levelType value valueQualitative valueAxisX valueAxisY valueType x y currency }
      scoreCoordinates { x y }
      scoreValueList { type levelIndex levelKey levelType metricId min max customName value valueQualitative valueAxisX valueAxisY valueSourceList valueType x y currency schemaId sourceId sourceType sourceEvaluationId sourceList { evaluationId sourceId sourceType }}
      dateValidFrom dateValidTo distribution distributionType type schemaId authorId metadata treatmentStrategy respondentIdList methodIdList createdAt updatedAt
    }
  }
  `,
  variables: {
    input: {
      filter: {
        parentEntityIdList: {
          $raw: `this && array['ECRS1YCtGEXx']::varchar[]`,
        },
      },
    },
  },
});

const targetEvaluationIdList = evaluationList.filter(x => x.type === 'target').map(x => x.id);

// Vraci Opatreni pro hodnoceni
const measureList = await getGraphql({
  query: `query measureList($input: ConnectionInputType) {
    measureListQuery(input: $input) {
      id key state category typeList authorId sourceId sourceTypeId targetId targetTypeId relatedData    
      evaluationSchemaId evaluationCurrentId evaluationTargetId evaluationActualId costs score validFrom validTo parentId metadata name description cybersecurityConcepts informationSecurityProperties operationalCapabilities securityDomains number importance isTemplate isWrapper isRequired methodIdList createdAt updatedAt
      measureHasEntityList { id state type measureId entityId entityTypeId entityFragmentId metadata authorId createdAt updatedAt }
    }
  }`,
  variables: {
    input: {
      filter: {
        evaluationTargetId: targetEvaluationIdList,
      },
    },
  },
});

const result = {
  evaluationList,
  measureList,
};
