const groupList = await getGraphql({
  query: `query getGroupList($input: GroupQueryType) {
    groupListQuery(input: $input) {
      id, name, metadata, entityList { id }
    }
  }`,
  variables: {
    input: {
      filter: {
        groupSchemaId: 'organization',
      },
    },
  },
});

const groupMap = groupList.reduce((map, group) => {
  map[group.id] = group;
  return map;
}, {});
